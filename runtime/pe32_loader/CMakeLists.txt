cmake_minimum_required(VERSION 3.20)
project(battle_realms_pe32_loader LANGUAGES CXX)

add_library(br_pe32_image STATIC ../pe32_image.cpp)
target_include_directories(br_pe32_image PUBLIC ..)
target_compile_features(br_pe32_image PUBLIC cxx_std_17)

add_library(br_guest_address_space STATIC ../guest_address_space.cpp)
target_include_directories(br_guest_address_space PUBLIC ..)
target_link_libraries(br_guest_address_space PUBLIC br_pe32_image)
target_compile_features(br_guest_address_space PUBLIC cxx_std_17)

add_library(br_ra3_guest_bridge STATIC ../ra3_guest_bridge.cpp)
target_include_directories(br_ra3_guest_bridge PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_ra3_guest_bridge PUBLIC br_guest_address_space)
target_compile_features(br_ra3_guest_bridge PUBLIC cxx_std_17)

add_library(br_import_binder STATIC ../import_binder.cpp)
target_include_directories(br_import_binder PUBLIC ..)
target_link_libraries(br_import_binder PUBLIC br_guest_address_space br_pe32_image)
target_compile_features(br_import_binder PUBLIC cxx_std_17)

add_library(br_lift_dispatch STATIC ../lift_dispatch.cpp)
target_include_directories(br_lift_dispatch PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_lift_dispatch PUBLIC br_import_binder br_ra3_guest_bridge)
target_compile_features(br_lift_dispatch PUBLIC cxx_std_17)
if(ANDROID AND BR_ANDROID_GUEST_PROFILE)
	# Keep profile state in the dispatcher archive so imported Win32/COM handlers
	# can be excluded from guest-function residency samples.
	target_sources(br_lift_dispatch PRIVATE ../android_guest_profiler.cpp)
	target_compile_definitions(br_lift_dispatch PRIVATE BR_ANDROID_GUEST_PROFILE=1)
endif()

add_library(br_win32_base STATIC ../win32_base.cpp)
target_include_directories(br_win32_base PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_win32_base PUBLIC br_lift_dispatch)
target_compile_features(br_win32_base PUBLIC cxx_std_17)

add_library(br_win32_startup STATIC ../win32_startup.cpp)
if(ANDROID)
	# Keep the established host software renderer as the reference. Android uses
	# an offscreen GLES backend for supported fixed-function draws and falls back
	# to that same software path whenever a state/format is not implemented.
	target_sources(br_win32_startup PRIVATE ../d3d7_android_gles_backend.cpp)
	target_compile_definitions(br_win32_startup PRIVATE BR_ANDROID_D3D7_GLES=1)
	if(BR_ANDROID_D3D7_TEXTURE_DIAGNOSTICS)
		target_compile_definitions(br_win32_startup PRIVATE BR_ANDROID_D3D7_TEXTURE_DIAGNOSTICS=1)
	endif()
	target_link_libraries(br_win32_startup PUBLIC EGL GLESv3)
else()
	target_sources(br_win32_startup PRIVATE ../d3d7_gpu_backend_stub.cpp)
endif()
target_include_directories(br_win32_startup PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_win32_startup PUBLIC br_win32_base br_lift_dispatch)
target_compile_features(br_win32_startup PUBLIC cxx_std_17)

add_library(br_game_runtime STATIC ../game_runtime.cpp)
target_include_directories(br_game_runtime PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_game_runtime PUBLIC br_import_binder br_lift_dispatch br_ra3_guest_bridge br_win32_startup)
target_compile_features(br_game_runtime PUBLIC cxx_std_17)

add_library(br_win32_sync STATIC ../win32_sync.cpp)
target_include_directories(br_win32_sync PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_win32_sync PUBLIC br_win32_base br_lift_dispatch)
target_compile_features(br_win32_sync PUBLIC cxx_std_17)

add_library(br_win32_memory STATIC ../win32_memory.cpp)
target_include_directories(br_win32_memory PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_win32_memory PUBLIC br_win32_base br_lift_dispatch)
target_compile_features(br_win32_memory PUBLIC cxx_std_17)

add_library(br_win32_file STATIC ../win32_file.cpp)
target_include_directories(br_win32_file PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
target_link_libraries(br_win32_file PUBLIC br_win32_base br_lift_dispatch)
target_compile_features(br_win32_file PUBLIC cxx_std_17)

add_executable(pe32_loader_smoke ../../tools/pe32_loader_smoke.cpp)
target_link_libraries(pe32_loader_smoke PRIVATE br_pe32_image br_guest_address_space br_import_binder)
target_compile_features(pe32_loader_smoke PRIVATE cxx_std_17)

if(NOT ANDROID)
	add_executable(android_guest_profile_scope_smoke ../../tools/android_guest_profile_scope_smoke.cpp)
	target_include_directories(android_guest_profile_scope_smoke PRIVATE ..)
	target_compile_features(android_guest_profile_scope_smoke PRIVATE cxx_std_17)

	add_executable(game_runtime_smoke ../../tools/game_runtime_smoke.cpp)
	target_include_directories(game_runtime_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
	target_link_libraries(game_runtime_smoke PRIVATE br_game_runtime br_win32_startup br_win32_file br_win32_memory br_win32_sync br_win32_base)
	target_compile_features(game_runtime_smoke PRIVATE cxx_std_17)
	if(WIN32)
		# Exercise the Android SAF-backed drive enumeration policy on the host:
		# C: is exposed only when the configured opener can read the game EXE.
		add_library(br_win32_file_virtual_drive_smoke STATIC ../win32_file.cpp)
		target_include_directories(br_win32_file_virtual_drive_smoke PUBLIC .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
		target_link_libraries(br_win32_file_virtual_drive_smoke PUBLIC br_win32_base br_lift_dispatch)
		target_compile_features(br_win32_file_virtual_drive_smoke PUBLIC cxx_std_17)
		target_compile_definitions(br_win32_file_virtual_drive_smoke PRIVATE BR_FORCE_VIRTUAL_DRIVE_ENUMERATION=1)
		add_executable(game_runtime_virtual_drive_smoke ../../tools/game_runtime_smoke.cpp)
		target_include_directories(game_runtime_virtual_drive_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
		target_compile_definitions(game_runtime_virtual_drive_smoke PRIVATE BR_TEST_VIRTUAL_DRIVE_ENUMERATION=1)
		target_link_libraries(game_runtime_virtual_drive_smoke PRIVATE br_game_runtime br_win32_startup br_win32_file_virtual_drive_smoke br_win32_memory br_win32_sync br_win32_base)
		target_compile_features(game_runtime_virtual_drive_smoke PRIVATE cxx_std_17)
	endif()

	add_executable(guest_bridge_smoke ../../tools/guest_bridge_smoke.cpp)
	target_include_directories(guest_bridge_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
	target_link_libraries(guest_bridge_smoke PRIVATE br_ra3_guest_bridge)
	target_compile_features(guest_bridge_smoke PRIVATE cxx_std_17)

	add_executable(lift_dispatch_smoke ../../tools/lift_dispatch_smoke.cpp)
	target_include_directories(lift_dispatch_smoke PRIVATE .. "${CMAKE_CURRENT_SOURCE_DIR}/../../references/RA3-Native/runtime")
	target_link_libraries(lift_dispatch_smoke PRIVATE br_win32_file br_win32_memory br_win32_sync br_win32_base br_lift_dispatch)
	target_compile_features(lift_dispatch_smoke PRIVATE cxx_std_17)
endif()

if(MSVC)
	foreach(target br_pe32_image br_guest_address_space br_ra3_guest_bridge br_import_binder br_game_runtime br_lift_dispatch br_win32_base br_win32_startup br_win32_sync br_win32_memory br_win32_file pe32_loader_smoke)
		target_compile_options(${target} PRIVATE /W4 /permissive-)
	endforeach()
	foreach(target android_guest_profile_scope_smoke game_runtime_smoke game_runtime_virtual_drive_smoke br_win32_file_virtual_drive_smoke guest_bridge_smoke lift_dispatch_smoke)
		if(TARGET ${target})
			target_compile_options(${target} PRIVATE /W4 /permissive-)
		endif()
	endforeach()
else()
	foreach(target br_pe32_image br_guest_address_space br_ra3_guest_bridge br_import_binder br_game_runtime br_lift_dispatch br_win32_base br_win32_startup br_win32_sync br_win32_memory br_win32_file pe32_loader_smoke)
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
	endforeach()
		foreach(target android_guest_profile_scope_smoke game_runtime_smoke game_runtime_virtual_drive_smoke br_win32_file_virtual_drive_smoke guest_bridge_smoke lift_dispatch_smoke)
		if(TARGET ${target})
			target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
		endif()
	endforeach()
endif()
