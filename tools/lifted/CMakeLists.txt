cmake_minimum_required(VERSION 3.22)
project(battle_realms_lifted CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT RA3_RUNTIME_DIR)
  message(FATAL_ERROR "Set RA3_RUNTIME_DIR")
endif()
if(LIFTED_DIRS)
  set(BR_LIFTED_DIRS ${LIFTED_DIRS})
elseif(LIFTED_DIR)
  set(BR_LIFTED_DIRS ${LIFTED_DIR})
else()
  message(FATAL_ERROR "Set LIFTED_DIR or LIFTED_DIRS")
endif()

set(LIFTED_SOURCES)
set(LIFTED_INCLUDE_DIRS)
foreach(LIFTED_SOURCE_DIR IN LISTS BR_LIFTED_DIRS)
  file(TO_CMAKE_PATH "${LIFTED_SOURCE_DIR}" LIFTED_SOURCE_DIR_CMAKE)
  file(GLOB LIFTED_TOP_LEVEL CONFIGURE_DEPENDS "${LIFTED_SOURCE_DIR_CMAKE}/lifted_*.cpp")
  file(GLOB_RECURSE LIFTED_NESTED CONFIGURE_DEPENDS "${LIFTED_SOURCE_DIR_CMAKE}/*/lifted_*.cpp")
  if(BR_COMBINED_TABLE)
    list(FILTER LIFTED_TOP_LEVEL EXCLUDE REGEX "(^|/)lifted_table\\.cpp$")
    list(FILTER LIFTED_NESTED EXCLUDE REGEX "(^|/)lifted_table\\.cpp$")
  endif()
  list(APPEND LIFTED_SOURCES ${LIFTED_TOP_LEVEL} ${LIFTED_NESTED})
  list(APPEND LIFTED_INCLUDE_DIRS "${LIFTED_SOURCE_DIR_CMAKE}")
endforeach()
if(BR_LIFTED_EXCLUDE_SOURCES)
  list(REMOVE_ITEM LIFTED_SOURCES ${BR_LIFTED_EXCLUDE_SOURCES})
endif()
if(BR_LIFTED_EXTRA_SOURCES)
  foreach(BR_EXTRA_SOURCE IN LISTS BR_LIFTED_EXTRA_SOURCES)
    if(NOT EXISTS "${BR_EXTRA_SOURCE}")
      message(FATAL_ERROR "Diagnostic/override lifted source does not exist: ${BR_EXTRA_SOURCE}")
    endif()
    list(APPEND LIFTED_SOURCES "${BR_EXTRA_SOURCE}")
    get_filename_component(BR_EXTRA_INCLUDE_DIR "${BR_EXTRA_SOURCE}" DIRECTORY)
    list(APPEND LIFTED_INCLUDE_DIRS "${BR_EXTRA_INCLUDE_DIR}")
  endforeach()
endif()
if(BR_COMBINED_TABLE)
  if(NOT EXISTS "${BR_COMBINED_TABLE}")
    message(FATAL_ERROR "BR_COMBINED_TABLE does not exist: ${BR_COMBINED_TABLE}")
  endif()
  list(APPEND LIFTED_SOURCES "${BR_COMBINED_TABLE}")
  if(BR_GENERATED_ROOT)
    list(APPEND LIFTED_INCLUDE_DIRS "${BR_GENERATED_ROOT}")
  endif()
endif()
if(NOT LIFTED_SOURCES)
  message(FATAL_ERROR "No lifted_*.cpp sources in ${BR_LIFTED_DIRS}")
endif()
add_library(battle_realms_lifted OBJECT ${LIFTED_SOURCES})
target_include_directories(battle_realms_lifted PRIVATE ${LIFTED_INCLUDE_DIRS}
	"${RA3_RUNTIME_DIR}" "${CMAKE_CURRENT_LIST_DIR}/../../runtime")
target_compile_options(battle_realms_lifted PRIVATE -O2 -fno-strict-aliasing -w -fno-emulated-tls)

option(BR_TRACE_LIFTED_FUNCTIONS "Compile per-function guest call tracing into lifted code" OFF)
if(BR_TRACE_LIFTED_FUNCTIONS)
  target_compile_definitions(battle_realms_lifted PRIVATE RA3_TRACE)
endif()

option(BR_PROFILE_GUEST_FUNCTION_SCOPE "Track current lifted function for Android sampling" OFF)
if(BR_PROFILE_GUEST_FUNCTION_SCOPE)
  if(NOT BR_TRACE_LIFTED_FUNCTIONS)
    message(FATAL_ERROR "BR_PROFILE_GUEST_FUNCTION_SCOPE requires BR_TRACE_LIFTED_FUNCTIONS=ON")
  endif()
  target_compile_options(battle_realms_lifted PRIVATE
    "-include" "${CMAKE_CURRENT_LIST_DIR}/../../runtime/android_guest_profile_trace.h")
endif()

option(BR_USE_CCACHE "Use ccache when installed" OFF)
if(BR_USE_CCACHE)
  find_program(BR_CCACHE ccache)
  if(BR_CCACHE)
    set_property(TARGET battle_realms_lifted PROPERTY CXX_COMPILER_LAUNCHER "${BR_CCACHE}")
  else()
    message(WARNING "BR_USE_CCACHE requested, but ccache was not found; Ninja incremental builds remain enabled")
  endif()
endif()
